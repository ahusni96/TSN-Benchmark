###	Chapter 4 - Time Aware Shaper (TAS)


==================== Start sync
A-side:~$ ptp4l -P2Hi enp1s0 -f ~/gPTP.cfg --step_threshold=1 -m --first_step_threshold=0.0
B-side:~$ ptp4l -P2Hi enp1s0 -f ~/gPTP.cfg --step_threshold=1 -m --first_step_threshold=0.0
A-side:~$ phc2sys -s enp1s0 -O 0 -c CLOCK_REALTIME --step_threshold=1 --transportSpecific=1 -w -m --first_step_threshold=0.0
B-side:~$ phc2sys -s enp1s0 -O 0 -c CLOCK_REALTIME --step_threshold=1 --transportSpecific=1 -w -m --first_step_threshold=0.0

==================== Clear qdisc
A-side:~$ tc qdisc del dev enp1s0 root
B-side:~$ tc qdisc del dev enp1s0 root

==================== Setup traffic
A-side:~$ BASE=$(date +%s%N) && sleep 1 && echo "$BASE"
A-side:~$ tc qdisc add dev enp1s0 parent root stab overhead 24 handle 100 taprio num_tc 4 map 0 1 2 3 3 3 3 3 3 3 3 3 3 3 3 3 queues 1@0 1@1 1@2 1@3 base-time 883618029032270886 sched-entry S 01 5000000 sched-entry S 02 5000000 sched-entry S 04 5000000 sched-entry S 08 5000000 flags 0x2 txtime-delay 0

==================== Setup net_prio
A-side:~$ mkdir -p /sys/fs/cgroup/net_prio
A-side:~$ mount -t cgroup -onet_prio none /sys/fs/cgroup/net_prio
A-side:~$ mkdir -p /sys/fs/cgroup/net_prio/grp1 && mkdir -p /sys/fs/cgroup/net_prio/grp2 && mkdir -p /sys/fs/cgroup/net_prio/grp3 && mkdir -p /sys/fs/cgroup/net_prio/grp4
A-side:~$ echo "enp1s0 1" > /sys/fs/cgroup/net_prio/grp1/net_prio.ifpriomap && echo "enp1s0 2" > /sys/fs/cgroup/net_prio/grp2/net_prio.ifpriomap && echo "enp1s0 3" > /sys/fs/cgroup/net_prio/grp3/net_prio.ifpriomap && echo "enp1s0 4" > /sys/fs/cgroup/net_prio/grp4/net_prio.ifpriomap

==================== Set IP address
A-side:~$ ifconfig enp1s0 169.254.1.11
A-side:~$ ip -4 addr show enp1s0 | grep -oP "(?<=inet\s)\d+(\.\d+){3}"
IP Address: 169.254.1.11

B-side:~$ ifconfig enp1s0 169.254.1.22
B-side:~$ ip -4 addr show enp1s0 | grep -oP "(?<=inet\s)\d+(\.\d+){3}"
IP Address: 169.254.1.22

==================== Start IPERF3 client
B-side:~$ iperf3 -s -p 5100 & iperf3 -s -p 5101 & iperf3 -s -p 5102 & iperf3 -s -p 5103

==================== Start IPERF3 server
A-side:~$ iperf3 -u -l1000 -b0 -p 5100 -c 169.254.1.22 -t 150 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp1/cgroup.procs && iperf3 -u -l1000 -b0 -p 5101 -c 169.254.1.22 -t 150 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp2/cgroup.procs && iperf3 -u -l1000 -b0 -p 5102 -c 169.254.1.22 -t 150 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp3/cgroup.procs

==================== Start TCPDUMP
B-side:~$ tcpdump -Q in -nvvvXXi enp1s0 --time-stamp-precision=nano -j adapter_unsynced -B 524288 -s 54 -w chp4_script.pcap -c 1000000
