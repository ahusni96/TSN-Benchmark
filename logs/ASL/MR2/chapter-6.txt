
==================== Start sync
A-side:~$ ptp4l -P2Hi enp0s30f4 -f ~/gPTP.cfg --step_threshold=1 -m
B-side:~$ ptp4l -P2Hi enp0s30f4 -f ~/gPTP.cfg --step_threshold=1 -m
A-side:~$ phc2sys -s enp0s30f4 -O 0 -c CLOCK_REALTIME --step_threshold=1 --transportSpecific=1 -w -m
B-side:~$ phc2sys -s enp0s30f4 -O 0 -c CLOCK_REALTIME --step_threshold=1 --transportSpecific=1 -w -m

==================== Clear qdisc
A-side:~$ tc qdisc del dev enp0s30f4 root
B-side:~$ tc qdisc del dev enp0s30f4 root

==================== Add qdisc
A-side:~$ BASE=$(expr $(date +%s) + 5)000000000 && tc qdisc add dev enp0s30f4 parent root stab overhead 24 handle 100 taprio num_tc 4 map 0 1 2 3 3 3 3 3 3 3 3 3 3 3 3 queues 1@0 1@1 1@2 1@3 base-time $BASE sched-entry R 08 5000000 sched-entry H 00 5000000 sched-entry R 08 5000000 sched-entry H 00 5000000 flags 0x2 txtime-delay 0 preempt 0111


==================== Show Frame Preemption
A-side:~$ dmesg | grep FPE | grep -ioPm1 "(?<=pec = )[^ ]*"
FPE: TxQ PEC = 0x7

==================== Stop sync
A-side:~$ pkill phc2sys
A-side:~$ pkill ptp4l
B-side:~$ pkill phc2sys
B-side:~$ pkill ptp4l
