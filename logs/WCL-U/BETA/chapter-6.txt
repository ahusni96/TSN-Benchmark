###	Chapter 6 - Qbu Frame Preemption

==================== Restart NetworkManager
A-side:~$ systemctl restart NetworkManager
B-side:~$ systemctl restart NetworkManager

==================== Clear qdisc
A-side:~$ tc qdisc del dev enp2s0 root
B-side:~$ tc qdisc del dev enp2s0 root

==================== Enable FPE (ETHTOOL)
A-side:~$ ethtool --set-mm enp2s0 pmac-enabled on tx-enabled on
B-side:~$ ethtool --set-mm enp2s0 pmac-enabled on tx-enabled on

==================== Verify Frame Preempt
A-side:~$ ethtool --set-mm enp2s0 verify-enabled on
B-side:~$ ethtool --set-mm enp2s0 verify-enabled on

==================== Configure TXQ PRIO
A-side:~$ ethtool --set-priv-flags enp2s0 reverse-tsn-txq-prio on
A-side:~$ ethtool --show-priv-flags enp2s0
Private flags for enp2s0:
legacy-rx           : off
reverse-tsn-txq-prio: on

==================== Set IP address
A-side:~$ ifconfig enp2s0 192.168.1.11
A-side:~$ ip -4 addr show enp2s0 | grep -oP "(?<=inet\s)\d+(\.\d+){3}"
IP Address: 192.168.1.11

B-side:~$ ifconfig enp2s0 192.168.1.22
B-side:~$ ip -4 addr show enp2s0 | grep -oP "(?<=inet\s)\d+(\.\d+){3}"
IP Address: 192.168.1.22

==================== Configure TAPRIO queue
A-side:~$ tc qdisc add dev enp2s0 parent root handle 100 taprio num_tc 4 map 0 1 2 3 3 3 3 3 3 3 3 3 3 3 3 3 queues 1@0 1@1 1@2 1@3 base-time $(date +%s%N) sched-entry S 0F 5000000 flags 0x2 fp P P E E

==================== Setup cgroup/net_prio folders
A-side:~$ mkdir -p /sys/fs/cgroup/net_prio
A-side:~$ mount -t cgroup -onet_prio none /sys/fs/cgroup/net_prio
A-side:~$ mkdir -p /sys/fs/cgroup/net_prio/grp1 && mkdir -p /sys/fs/cgroup/net_prio/grp2 && mkdir -p /sys/fs/cgroup/net_prio/grp3 && mkdir -p /sys/fs/cgroup/net_prio/grp4
A-side:~$ echo "enp2s0 1" > /sys/fs/cgroup/net_prio/grp1/net_prio.ifpriomap && echo "enp2s0 2" > /sys/fs/cgroup/net_prio/grp2/net_prio.ifpriomap && echo "enp2s0 3" > /sys/fs/cgroup/net_prio/grp3/net_prio.ifpriomap && echo "enp2s0 4" > /sys/fs/cgroup/net_prio/grp4/net_prio.ifpriomap

==================== Send traffic
B-side:~$ iperf3 -s -p 5100 > ~/5100.txt & iperf3 -s -p 5101 > ~/5101.txt & iperf3 -s -p 5102 > ~/5102.txt & iperf3 -s -p 5103 > ~/5103.txt
A-side:~$ iperf3 -u -l 1250 -b0 -p 5100 -c 192.168.1.22 -t 30 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp1/cgroup.procs && iperf3 -u -l 1250 -b0 -p 5101 -c 192.168.1.22 -t 30>> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp2/cgroup.procs && iperf3 -u -l 68 -b0 -p 5102 -c 192.168.1.22 -t 30 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp3/cgroup.procs && iperf3 -u -l 68 -b0 -p 5103 -c 192.168.1.22 -t 30 >> /dev/null & echo $! > /sys/fs/cgroup/net_prio/grp4/cgroup.procs
